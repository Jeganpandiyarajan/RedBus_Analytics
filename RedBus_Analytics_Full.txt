/* ============================================================
   STEP 4: DIM_DATE - Calendar Dimension
   ============================================================ */

USE SCHEMA DW;

CREATE OR REPLACE TABLE DW.DIM_DATE (
  SK_DATE       NUMBER PRIMARY KEY,
  DATE_VALUE    DATE,
  YEAR          NUMBER,
  QUARTER       NUMBER,
  MONTH         NUMBER,
  MONTH_NAME    STRING,
  WEEK_OF_YEAR  NUMBER,
  DAY_OF_WEEK   NUMBER,
  DAY_NAME      STRING,
  IS_WEEKEND    BOOLEAN
);

INSERT INTO DW.DIM_DATE
SELECT
  TO_NUMBER(TO_CHAR(D, 'YYYYMMDD')) AS SK_DATE,
  D AS DATE_VALUE,
  YEAR(D) AS YEAR,
  QUARTER(D) AS QUARTER,
  MONTH(D) AS MONTH,
  MONTHNAME(D) AS MONTH_NAME,
  WEEKOFYEAR(D) AS WEEK_OF_YEAR,
  DAYOFWEEK(D) AS DAY_OF_WEEK,
  DAYNAME(D) AS DAY_NAME,
  CASE WHEN DAYOFWEEK(D) IN (0,6) THEN TRUE ELSE FALSE END AS IS_WEEKEND
FROM (
  SELECT DATEADD(DAY, SEQ4(), '2020-01-01') AS D
  FROM TABLE(GENERATOR(ROWCOUNT => 2922))
);


/* ============================================================
   STEP 5: DIM_ROUTE - Type 1 Dimension
   ============================================================ */

CREATE OR REPLACE TABLE DW.DIM_ROUTE (
  SK_ROUTE        NUMBER AUTOINCREMENT START 1 INCREMENT 1 PRIMARY KEY,
  ROUTE_ID        STRING,
  SOURCE_CITY     STRING,
  DEST_CITY       STRING,
  DISTANCE_KM     NUMBER,
  SOURCE_STATE    STRING,
  DEST_STATE      STRING,
  REGION          STRING,
  EFFECTIVE_FROM  DATE,
  STATUS          STRING,
  LOAD_TS         TIMESTAMP_NTZ DEFAULT CURRENT_TIMESTAMP()
);

MERGE INTO DW.DIM_ROUTE AS TGT
USING RAW.RAW_ROUTES AS SRC
ON TGT.ROUTE_ID = SRC.ROUTE_ID
WHEN MATCHED THEN UPDATE SET
  SOURCE_CITY     = SRC.SOURCE_CITY,
  DEST_CITY       = SRC.DEST_CITY,
  DISTANCE_KM     = SRC.DISTANCE_KM,
  SOURCE_STATE    = SRC.SOURCE_STATE,
  DEST_STATE      = SRC.DEST_STATE,
  REGION          = SRC.REGION,
  EFFECTIVE_FROM  = SRC.EFFECTIVE_FROM,
  STATUS          = SRC.STATUS,
  LOAD_TS         = CURRENT_TIMESTAMP()
WHEN NOT MATCHED THEN INSERT (
  ROUTE_ID, SOURCE_CITY, DEST_CITY, DISTANCE_KM,
  SOURCE_STATE, DEST_STATE, REGION, EFFECTIVE_FROM, STATUS, LOAD_TS
) VALUES (
  SRC.ROUTE_ID, SRC.SOURCE_CITY, SRC.DEST_CITY, SRC.DISTANCE_KM,
  SRC.SOURCE_STATE, SRC.DEST_STATE, SRC.REGION,
  SRC.EFFECTIVE_FROM, SRC.STATUS, CURRENT_TIMESTAMP()
);

INSERT INTO DW.DIM_CUSTOMER (
  CUSTOMER_ID, FIRST_NAME, LAST_NAME, EMAIL, PHONE,
  SEGMENT, CITY, STATE, COUNTRY,
  HASH_DIFF, EFF_START_TS, EFF_END_TS, IS_CURRENT

  CUSTOMER_ID, FIRST_NAME, LAST_NAME, EMAIL, PHONE,
  SEGMENT, CITY, STATE, COUNTRY,

                SEGMENT, CITY, STATE, COUNTRY)) AS HASH_DIFF,
  CURRENT_TIMESTAMP() AS EFF_START_TS,
  '9999-12-31'::TIMESTAMP_NTZ AS EFF_END_TS,
  TRUE AS IS_CURRENT
FROM RAW.RAW_CUSTOMERS;

/* ============================================================
   STEP 8: FACT_BOOKING - Fact Table
   Grain: One row per booking line (BOOKING_ID + BOOKING_LINE_ID)
   ============================================================ */

CREATE OR REPLACE TABLE DW.FACT_BOOKING (
  BOOKING_ID        STRING,
  BOOKING_LINE_ID   NUMBER,
  SK_BOOKING_DATE   NUMBER,
  SK_JOURNEY_DATE   NUMBER,
  SK_CUSTOMER       NUMBER,
  SK_ROUTE          NUMBER,
  SK_BUS            NUMBER,
  DEPARTURE_TIME    STRING,
  ARRIVAL_TIME      STRING,
  SEAT_TYPE         STRING,
  SEAT_NO           STRING,
  PASSENGER_COUNT   NUMBER,
  FARE_AMOUNT       NUMBER(10,2),
  DISCOUNT_PCT      NUMBER(5,2),
  NET_AMOUNT        NUMBER(10,2),
  PAYMENT_MODE      STRING,
  BOOKING_STATUS    STRING,
  LOAD_TS           TIMESTAMP_NTZ DEFAULT CURRENT_TIMESTAMP(),
  PRIMARY KEY (BOOKING_ID, BOOKING_LINE_ID)
);

INSERT INTO DW.FACT_BOOKING (
  BOOKING_ID, BOOKING_LINE_ID,
  SK_BOOKING_DATE, SK_JOURNEY_DATE,
  SK_CUSTOMER, SK_ROUTE, SK_BUS,
  DEPARTURE_TIME, ARRIVAL_TIME, SEAT_TYPE, SEAT_NO,
  PASSENGER_COUNT, FARE_AMOUNT, DISCOUNT_PCT, NET_AMOUNT,
  PAYMENT_MODE, BOOKING_STATUS
)
SELECT
  B.BOOKING_ID,
  B.BOOKING_LINE_ID,
  TO_NUMBER(TO_CHAR(B.BOOKING_DATE, 'YYYYMMDD')) AS SK_BOOKING_DATE,
  TO_NUMBER(TO_CHAR(B.JOURNEY_DATE, 'YYYYMMDD')) AS SK_JOURNEY_DATE,
  COALESCE(C.SK_CUSTOMER, -1) AS SK_CUSTOMER,
  COALESCE(R.SK_ROUTE, -1) AS SK_ROUTE,
  COALESCE(BU.SK_BUS, -1) AS SK_BUS,
  B.DEPARTURE_TIME,
  B.ARRIVAL_TIME,
  B.SEAT_TYPE,
  B.SEAT_NO,
  B.PASSENGER_COUNT,
  B.FARE_AMOUNT,
  B.DISCOUNT_PCT,
  ROUND(B.FARE_AMOUNT * B.PASSENGER_COUNT * (1 - B.DISCOUNT_PCT/100), 2) AS NET_AMOUNT,
  B.PAYMENT_MODE,
  B.BOOKING_STATUS
FROM RAW.RAW_BOOKINGS B
LEFT JOIN DW.DIM_CUSTOMER C ON B.CUSTOMER_ID = C.CUSTOMER_ID AND C.IS_CURRENT = TRUE
LEFT JOIN DW.DIM_ROUTE R ON B.ROUTE_ID = R.ROUTE_ID
LEFT JOIN DW.DIM_BUS BU ON B.BUS_ID = BU.BUS_ID AND BU.IS_CURRENT = TRUE;

-- Verify fact table row count
SELECT COUNT(*) AS FACT_ROW_COUNT FROM DW.FACT_BOOKING;

-- Check for unmatched dimension keys (should return 0)
SELECT COUNT(*) AS UNMATCHED_KEYS
FROM DW.FACT_BOOKING
WHERE SK_CUSTOMER = -1 OR SK_ROUTE = -1 OR SK_BUS = -1;

UNION ALL SELECT 'FACT_BOOKING', COUNT(*), 'SUCCESS', 'Initial load' FROM DW.FACT_BOOKING;

-- Reconciliation: RAW vs DW counts
SELECT
  (SELECT COUNT(*) FROM RAW.RAW_BOOKINGS) AS RAW_BOOKING_COUNT,
  (SELECT COUNT(*) FROM DW.FACT_BOOKING) AS DW_BOOKING_COUNT;

-- Business metric: Cancellation Rate
SELECT
  ROUND(100.0 * SUM(CASE WHEN BOOKING_STATUS = 'CANCELLED' THEN 1 ELSE 0 END) / COUNT(*), 2) AS CANCELLATION_RATE_PCT
FROM DW.FACT_BOOKING;

USE SCHEMA SEMANTIC;

-- Customer View
CREATE OR REPLACE VIEW SEMANTIC.VW_CUSTOMER AS
SELECT
  CUSTOMER_SK,
  CUSTOMER_ID,
  FIRST_NAME,
  LAST_NAME,
  CONCAT(FIRST_NAME, ' ', LAST_NAME) AS FULL_NAME,
  EMAIL,
  PHONE,
  SEGMENT,
  CITY,
  STATE,
  COUNTRY,
  CASE
    WHEN SEGMENT = 'PREMIUM' THEN 'High Value'
    WHEN SEGMENT = 'VIP' THEN 'Frequent Traveler'
    WHEN SEGMENT = 'REGULAR' THEN 'Standard'
    ELSE 'Unknown'
  END AS SEGMENT_DESCRIPTION,
  IS_CURRENT
FROM DW.DIM_CUSTOMER
WHERE IS_CURRENT = TRUE;

-- Bus View
CREATE OR REPLACE VIEW SEMANTIC.VW_BUS AS
SELECT
  BUS_SK,
  BUS_ID,
  OPERATOR_NAME,
  BUS_TYPE,
  SEAT_CAPACITY,
  BUS_MODEL,
  REGISTRATION_NO,
  IN_SERVICE_DATE,
  STATUS,
  CASE
    WHEN SEAT_CAPACITY > 40 THEN 'Large Bus'
    WHEN SEAT_CAPACITY > 30 THEN 'Medium Bus'
    WHEN SEAT_CAPACITY > 20 THEN 'Small Bus'
    ELSE 'Mini Bus'
  END AS BUS_SIZE_CATEGORY,
  IS_CURRENT
FROM DW.DIM_BUS
WHERE IS_CURRENT = TRUE;

-- Route View
CREATE OR REPLACE VIEW SEMANTIC.VW_ROUTE AS
SELECT
  SK_ROUTE,
  ROUTE_ID,
  SOURCE_CITY,
  DEST_CITY,
  CONCAT(SOURCE_CITY, ' -> ', DEST_CITY) AS ROUTE_DESCRIPTION,
  DISTANCE_KM,
  SOURCE_STATE,
  DEST_STATE,
  REGION,
  CASE
    WHEN DISTANCE_KM > 800 THEN 'Long Distance'
    WHEN DISTANCE_KM > 400 THEN 'Medium Distance'
    ELSE 'Short Distance'
  END AS DISTANCE_CATEGORY
FROM DW.DIM_ROUTE;

-- Date View
CREATE OR REPLACE VIEW SEMANTIC.VW_DATE AS
SELECT
  SK_DATE,
  DATE_VALUE AS FULL_DATE,
  YEAR,
  QUARTER,
  MONTH,
  MONTH_NAME,
  DAY_NAME,
  IS_WEEKEND,
  CASE
    WHEN IS_WEEKEND = TRUE THEN 'Weekend'
    ELSE 'Weekday'
  END AS DAY_TYPE,
  CASE
    WHEN MONTH >= 4 THEN YEAR
    ELSE YEAR - 1
  END AS FISCAL_YEAR
FROM DW.DIM_DATE;

-- Main Booking View (All dimensions joined)
CREATE OR REPLACE VIEW SEMANTIC.VW_BOOKINGS AS
SELECT
  F.BOOKING_ID,
  F.BOOKING_LINE_ID,
  F.SK_BOOKING_DATE,
  BD.FULL_DATE AS BOOKING_DATE,
  BD.YEAR AS BOOKING_YEAR,
  BD.MONTH_NAME AS BOOKING_MONTH,
  F.SK_JOURNEY_DATE,
  JD.FULL_DATE AS JOURNEY_DATE,
  JD.YEAR AS JOURNEY_YEAR,
  JD.MONTH_NAME AS JOURNEY_MONTH,
  JD.DAY_NAME AS JOURNEY_DAY,
  JD.IS_WEEKEND AS JOURNEY_IS_WEEKEND,
  F.SK_CUSTOMER,
  C.CUSTOMER_ID,
  C.FULL_NAME AS CUSTOMER_NAME,
  C.SEGMENT AS CUSTOMER_SEGMENT,
  C.CITY AS CUSTOMER_CITY,
  C.STATE AS CUSTOMER_STATE,
  C.SEGMENT_DESCRIPTION,
  F.SK_ROUTE,
  R.ROUTE_ID,
  R.SOURCE_CITY,
  R.DEST_CITY,
  R.ROUTE_DESCRIPTION,
  R.DISTANCE_KM,
  R.REGION,
  R.DISTANCE_CATEGORY,
  F.SK_BUS,
  BU.BUS_ID,
  BU.OPERATOR_NAME,
  BU.BUS_TYPE,
  BU.SEAT_CAPACITY,
  BU.BUS_SIZE_CATEGORY,
  F.DEPARTURE_TIME,
  F.ARRIVAL_TIME,
  F.SEAT_TYPE,
  F.SEAT_NO,
  F.PASSENGER_COUNT,
  F.FARE_AMOUNT,
  F.DISCOUNT_PCT,
  F.NET_AMOUNT,
  F.PAYMENT_MODE,
  F.BOOKING_STATUS,
  CASE
    WHEN F.BOOKING_STATUS = 'CANCELLED' THEN TRUE
    ELSE FALSE
  END AS IS_CANCELLED,
  F.FARE_AMOUNT / NULLIF(F.PASSENGER_COUNT, 0) AS FARE_PER_PASSENGER,
  F.NET_AMOUNT / NULLIF(F.PASSENGER_COUNT, 0) AS NET_PER_PASSENGER
FROM DW.FACT_BOOKING F
LEFT JOIN SEMANTIC.VW_DATE BD ON F.SK_BOOKING_DATE = BD.SK_DATE
LEFT JOIN SEMANTIC.VW_DATE JD ON F.SK_JOURNEY_DATE = JD.SK_DATE
LEFT JOIN SEMANTIC.VW_CUSTOMER C ON F.SK_CUSTOMER = C.CUSTOMER_SK
LEFT JOIN SEMANTIC.VW_ROUTE R ON F.SK_ROUTE = R.SK_ROUTE
LEFT JOIN SEMANTIC.VW_BUS BU ON F.SK_BUS = BU.BUS_SK;

-- Keyword-match search function
CREATE OR REPLACE FUNCTION SEMANTIC.SEARCH_POLICY(QUESTION STRING)
RETURNS TABLE (DOC_ID STRING, TITLE STRING, CONTENT STRING, RELEVANCE NUMBER)
AS
$$
  SELECT
    DOC_ID,
    TITLE,
    CONTENT,
    (SELECT COUNT(*)
     FROM TABLE(SPLIT_TO_TABLE(LOWER(QUESTION), ' ')) W
     WHERE LENGTH(W.VALUE) > 3
       AND W.VALUE NOT IN ('what','when','where','which','does','their','with','from','have','this','that')
       AND CONTAINS(LOWER(CONTENT), W.VALUE)
    ) AS RELEVANCE
  FROM SEMANTIC.POLICY_DOCS
  ORDER BY RELEVANCE DESC
  LIMIT 1
$$;

-- KPI Summary
SELECT
  COUNT(DISTINCT BOOKING_ID) AS TOTAL_BOOKINGS,
  SUM(NET_AMOUNT) AS TOTAL_REVENUE,
  AVG(NET_AMOUNT) AS AVG_FARE,
  SUM(CASE WHEN BOOKING_STATUS = 'CANCELLED' THEN 1 ELSE 0 END) AS CANCELLED_BOOKINGS,
  ROUND(100.0 * SUM(CASE WHEN BOOKING_STATUS = 'CANCELLED' THEN 1 ELSE 0 END) / COUNT(*), 2) AS CANCEL_RATE
FROM SEMANTIC.VW_BOOKINGS;

-- Top 5 Routes by Revenue
SELECT
  ROUTE_DESCRIPTION,
  COUNT(*) AS BOOKINGS,
  SUM(NET_AMOUNT) AS REVENUE
FROM SEMANTIC.VW_BOOKINGS
WHERE BOOKING_STATUS != 'CANCELLED'
GROUP BY ROUTE_DESCRIPTION
ORDER BY REVENUE DESC
LIMIT 5;

-- Top 5 Operators by Revenue
SELECT
  OPERATOR_NAME,
  COUNT(*) AS BOOKINGS,
  SUM(NET_AMOUNT) AS REVENUE
FROM SEMANTIC.VW_BOOKINGS
WHERE BOOKING_STATUS != 'CANCELLED'
GROUP BY OPERATOR_NAME
ORDER BY REVENUE DESC
LIMIT 5;

-- Monthly Revenue Trend
SELECT
  JOURNEY_YEAR,
  JOURNEY_MONTH,
  COUNT(*) AS BOOKINGS,
  SUM(NET_AMOUNT) AS REVENUE
FROM SEMANTIC.VW_BOOKINGS
GROUP BY JOURNEY_YEAR, JOURNEY_MONTH
ORDER BY JOURNEY_YEAR, JOURNEY_MONTH;

-- Customer Segment Performance
SELECT
  CUSTOMER_SEGMENT,
  COUNT(*) AS BOOKINGS,
  SUM(NET_AMOUNT) AS REVENUE,
  AVG(NET_AMOUNT) AS AVG_SPEND
FROM SEMANTIC.VW_BOOKINGS
WHERE BOOKING_STATUS != 'CANCELLED'
GROUP BY CUSTOMER_SEGMENT
ORDER BY REVENUE DESC;

-- Weekend vs Weekday Analysis
SELECT
  CASE WHEN JOURNEY_IS_WEEKEND = TRUE THEN 'Weekend' ELSE 'Weekday' END AS DAY_TYPE,
  COUNT(*) AS BOOKINGS,
  SUM(NET_AMOUNT) AS REVENUE,
  AVG(NET_AMOUNT) AS AVG_FARE
FROM SEMANTIC.VW_BOOKINGS
WHERE BOOKING_STATUS != 'CANCELLED'
GROUP BY JOURNEY_IS_WEEKEND
ORDER BY JOURNEY_IS_WEEKEND;
